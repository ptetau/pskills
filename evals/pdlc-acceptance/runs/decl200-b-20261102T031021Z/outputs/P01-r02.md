use std::fs;
use std::path::PathBuf;

use todo::handle;
use todo::http::{Request, Response};
use todo::store::{Item, Store};

fn path(name: &str) -> PathBuf {
    let dir = std::env::temp_dir().join(format!("todo-acc-in0001-{}-{}", std::process::id(), name));
    fs::create_dir_all(&dir).unwrap();
    let file = dir.join("todo.txt");
    let _ = fs::remove_file(&file);
    file
}

fn get(store: &Store, target: &str) -> Response {
    handle(store, &Request::new("GET", target, ""))
}

fn add(store: &Store, text: &str) {
    let r = handle(store, &Request::new("POST", "/items", &format!("text={}", text)));
    assert_eq!(r.status, 303);
}

fn done(store: &Store, id: &str) -> Response {
    handle(store, &Request::new("POST", &format!("/items/{}/done", id), ""))
}

// The text of each li row of a page, in page order.
fn rows(body: &str) -> Vec<String> {
    body.split("<li>")
        .skip(1)
        .map(|part| part.split("</li>").next().unwrap().to_string())
        .collect()
}

fn row_of<'a>(rows: &'a [String], text: &str) -> &'a String {
    rows.iter().find(|r| r.contains(text)).unwrap_or_else(|| panic!("no row with {}", text))
}
