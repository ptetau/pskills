````python
#!/usr/bin/env python3
"""Helper for the visual review of CH-0001. It runs wordfreq on a sample file.

Usage, from the project root:
  python3 pdlc/changes/CH-0001-top-option/case.py [wordfreq arguments]

The script writes the sample files into pdlc/.tools/s/ (not committed), and runs
`python3 -m wordfreq` there, so that the arguments name the files as the spec does.
It prints the sample file that the arguments name, standard output, standard error,
and the exit code, so that each stream is easy to see.
"""
import os
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
SAMPLES = ROOT / "pdlc" / ".tools" / "s"
FILES = {"abc.txt": "b a b c a b\n", "fruit.txt": "pear apple fig\n", "empty.txt": ""}


def main(args):
    SAMPLES.mkdir(parents=True, exist_ok=True)
    for name, text in FILES.items():
        (SAMPLES / name).write_text(text)
    (SAMPLES / "missing.txt").unlink(missing_ok=True)

    for arg in args:
        if arg == "missing.txt":
            print("missing.txt: no such file")
        elif arg in FILES:
            print("%s: %s" % (arg, FILES[arg].strip() or "empty file"))

    env = dict(os.environ, PYTHONPATH=str(ROOT), PYTHONDONTWRITEBYTECODE="1")
    run = subprocess.run([sys.executable, "-m", "wordfreq", *args], cwd=SAMPLES, env=env,
                         capture_output=True, text=True)
    for label, text in (("stdout", run.stdout), ("stderr", run.stderr)):
        lines = text.splitlines()
        if not lines:
            print("%s: (empty)" % label)
        else:
            print("%s:" % label)
            for line in lines:
                print(line)
    print("exit code: %d" % run.returncode)


if __name__ == "__main__":
    main(sys.argv[1:])
````
