### pdlc/changes/CH-0001-top-option/wf.sh
````
#!/bin/sh
# Shows what a user sees when wordfreq runs: the sample file, standard output,
# standard error, and the exit code.
# Run it from the project root:
#   sh pdlc/changes/CH-0001-top-option/wf.sh [wordfreq arguments]
# The sample files are abc.txt (b a b c a b), fruit.txt (pear apple fig), and empty.txt.
# The file missing.txt does not exist.
root=$(cd "$(dirname "$0")/../../.." && pwd)
work=$(mktemp -d)
trap 'rm -rf "$work"' EXIT
mkdir "$work/files" "$work/streams"
printf 'b a b c a b\n' > "$work/files/abc.txt"
printf 'pear apple fig\n' > "$work/files/fruit.txt"
: > "$work/files/empty.txt"
cd "$work/files" || exit 1

for arg in "$@"; do
  if [ -f "$arg" ]; then
    if [ -s "$arg" ]; then
      echo "$arg holds: $(cat "$arg")"
    else
      echo "$arg is empty"
    fi
  fi
done

PYTHONPATH="$root" PYTHONDONTWRITEBYTECODE=1 python3 -m wordfreq "$@" \
  > "$work/streams/out" 2> "$work/streams/err"
code=$?

show() {
  if [ -s "$2" ]; then
    echo "$1:"
    sed 's/^/  /' "$2"
  else
    echo "$1: (empty)"
  fi
}
show "standard output" "$work/streams/out"
show "standard error" "$work/streams/err"
echo "exit code: $code"
````
