### proj/pdlc/changes/CH-0001-top-option/try.sh

```
#!/bin/sh
# Helper for the visual review of CH-0001 (top option). It shows one run of wordfreq.
#
# Run it from the project root:
#   sh pdlc/changes/CH-0001-top-option/try.sh TEXT [ARGUMENT...]
#
# TEXT is the content of the sample file notes.txt. The text "-" makes no sample file.
# The script runs "python3 -m wordfreq ARGUMENT..." in a scratch folder that holds
# notes.txt. Then it prints standard output, standard error, and the exit code, each on
# its own, so that the screen shows which stream got which line.
root=$(pwd)
scratch=$(mktemp -d)
trap 'rm -rf "$scratch"' EXIT
cd "$scratch" || exit 1
if [ "$1" = "-" ]; then
  echo "no sample file"
elif [ -z "$1" ]; then
  : > notes.txt
  echo "notes.txt is empty"
else
  printf '%s\n' "$1" > notes.txt
  echo "notes.txt holds: $1"
fi
[ $# -gt 0 ] && shift
PYTHONDONTWRITEBYTECODE=1 PYTHONPATH="$root" python3 -m wordfreq "$@" >out.txt 2>err.txt
code=$?
show() {
  if [ -s "$2" ]; then
    echo "$1:"
    sed 's/^/  /' "$2"
  else
    echo "$1: (empty)"
  fi
}
show "standard output" out.txt
show "standard error" err.txt
echo "exit code: $code"
```
