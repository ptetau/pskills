### pdlc/changes/CH-0001-top-option/show.sh
```sh
#!/bin/sh
# Helper for the visual review of CH-0001 (top option). Run it from the project root.
#
# Usage: sh pdlc/changes/CH-0001-top-option/show.sh TEXT ARG...
#
# TEXT is the content of the sample file notes.txt. Use - for no sample file.
# The helper runs `python3 -m wordfreq ARG...` in a scratch folder, so that ARG can name
# notes.txt. It then prints the standard output, the standard error, and the exit code.

root=$(pwd)
text=$1
shift
work=$(mktemp -d "${TMPDIR:-/tmp}/wordfreq-show.XXXXXX") || exit 1
trap 'rm -rf "$work"' EXIT
cd "$work" || exit 1

if [ "$text" != "-" ]; then
  printf '%s' "$text" > notes.txt
  if [ -n "$text" ]; then
    echo "notes.txt holds: $text"
  else
    echo "notes.txt holds: nothing (an empty file)"
  fi
fi

PYTHONPATH=$root python3 -m wordfreq "$@" > out.txt 2> err.txt
code=$?

show() {
  if [ -s "$2" ]; then
    echo "$1:"
    sed 's/^/  /' "$2"
  else
    echo "$1: (empty)"
  fi
}

show stdout out.txt
show stderr err.txt
echo "exit code: $code"
```
