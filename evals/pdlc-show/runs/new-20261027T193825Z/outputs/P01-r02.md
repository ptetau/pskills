"""Show one run of wordfreq for the visual review of CH-0001 (top option).

Usage, from the project root:
  python3 pdlc/changes/CH-0001-top-option/case.py TEXT [ARGUMENT...]

TEXT is the content of notes.txt. The value "-" means that no file exists.
The script runs `python3 -m wordfreq ARGUMENT...` in an empty folder, and prints the
content of the file, the command, standard output, standard error, and the exit code.
"""
import os
import shlex
import subprocess
import sys
import tempfile

ROOT = os.getcwd()


def block(name, text):
    if not text:
        return [f"{name}: (empty)"]
    return [f"{name}:"] + ["  " + line for line in text.splitlines()]


def main():
    text, arguments = sys.argv[1], sys.argv[2:]
    with tempfile.TemporaryDirectory() as folder:
        if text == "-":
            print("files: none")
        else:
            with open(os.path.join(folder, "notes.txt"), "w", encoding="utf-8") as handle:
                handle.write(text)
            print("notes.txt holds: " + (text if text else "(nothing)"))
        command = ["python3", "-m", "wordfreq"] + arguments
        print("command: " + shlex.join(command))
        env = dict(os.environ, PYTHONPATH=ROOT, PYTHONDONTWRITEBYTECODE="1")
        done = subprocess.run(command, cwd=folder, env=env, capture_output=True, text=True)
    for line in block("stdout", done.stdout) + block("stderr", done.stderr):
        print(line)
    print(f"exit code: {done.returncode}")


main()
