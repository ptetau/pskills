### pdlc/changes/CH-0001-top-option/wf.py

````python
"""Run wordfreq as a user does, and show what it did.

Usage: python3 pdlc/changes/CH-0001-top-option/wf.py [ARGS...]

The command runs `python3 -m wordfreq ARGS...` in the folder `samples/` next to this file.
It shows the standard output, the standard error, and the exit code on separate lines.
"""
import os
import subprocess
import sys

here = os.path.dirname(os.path.abspath(__file__))
root = os.path.abspath(os.path.join(here, "..", "..", ".."))
env = {**os.environ, "PYTHONPATH": root, "PYTHONDONTWRITEBYTECODE": "1"}
done = subprocess.run(
    [sys.executable, "-m", "wordfreq", *sys.argv[1:]],
    cwd=os.path.join(here, "samples"),
    env=env,
    capture_output=True,
    text=True,
)


def show(name, text):
    lines = text.splitlines()
    if not lines:
        print(name + ": (empty)")
        return
    print(name + ":")
    for line in lines:
        print("  " + line)


show("stdout", done.stdout)
show("stderr", done.stderr)
print("exit code: " + str(done.returncode))
````
