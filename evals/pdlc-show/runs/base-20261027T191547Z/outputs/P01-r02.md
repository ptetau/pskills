### proj/pdlc/changes/CH-0001-top-option/show-case.sh (new)
````sh
#!/bin/sh
# Shows one run of the wordfreq command for the visual review: the command, what it wrote
# to standard output and to standard error, and its exit code.
# Run it from the project root: sh pdlc/changes/CH-0001-top-option/show-case.sh --top 2 words.txt
# The command runs in the samples/ folder next to this script, so a file name stays short.
root=$(pwd)
here=$(cd "$(dirname "$0")" && pwd)
out=$(mktemp)
err=$(mktemp)
cd "$here/samples" || exit 1
PYTHONDONTWRITEBYTECODE=1 PYTHONPATH="$root" python3 -m wordfreq "$@" >"$out" 2>"$err"
code=$?
printf 'ran    python3 -m wordfreq'
for arg in "$@"; do
  case $arg in
    ''|*[!A-Za-z0-9._+-]*) printf " '%s'" "$arg" ;;
    *) printf ' %s' "$arg" ;;
  esac
done
printf '\n'
show() {
  if [ -s "$2" ]; then
    while IFS= read -r line; do printf '%s %s\n' "$1" "$line"; done <"$2"
  else
    printf '%s (empty)\n' "$1"
  fi
}
show stdout "$out"
show stderr "$err"
printf 'exit   %s\n' "$code"
rm -f "$out" "$err"
````
