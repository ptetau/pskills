### pdlc/changes/CH-0001-top-option/case.py
````python
"""Run `python3 -m wordfreq` for the visual review of CH-0001 and show the result.

Usage: python3 pdlc/changes/CH-0001-top-option/case.py [wordfreq arguments]

The command runs in the samples folder, so an argument such as abc.txt names a sample
file. The script prints what the sample files hold. Then it prints standard output,
standard error, and the exit code, each apart.
"""
import os
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[3]
samples = Path(__file__).resolve().parent / "samples"
args = sys.argv[1:]

for name in args:
    sample = samples / name
    if sample.is_file():
        words = sample.read_text(encoding="utf-8").strip() or "(nothing)"
        print("file " + name + " holds: " + words)

env = dict(os.environ, PYTHONPATH=str(root), PYTHONDONTWRITEBYTECODE="1")
done = subprocess.run(
    [sys.executable, "-m", "wordfreq", *args],
    cwd=samples, env=env, capture_output=True, text=True,
)
print("standard output:" + ("" if done.stdout else " (empty)"))
print(done.stdout, end="")
print("standard error:" + ("" if done.stderr else " (empty)"))
print(done.stderr, end="")
print("exit code: " + str(done.returncode))
````
