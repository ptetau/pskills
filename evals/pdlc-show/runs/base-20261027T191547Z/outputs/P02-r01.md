"""Run `python3 -m wordfreq` with the given arguments, for the visual review of CH-0001.

The arguments pass to the command unchanged. The command runs in the `samples` folder
next to this script. Thus a file name such as `letters.txt` finds a sample file.
The script prints the sample files that the arguments name. Then it prints the standard
output, the standard error, and the exit code of the command, each on its own line.
"""
import os
import shlex
import subprocess
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
SAMPLES = HERE / "samples"
ROOT = HERE.parents[2]


def show(label, text):
    lines = text.rstrip("\n").split("\n") if text else []
    if not lines:
        print(label + ": (empty)")
    elif len(lines) == 1:
        print(label + ": " + lines[0])
    else:
        print(label + ":")
        for line in lines:
            print("  " + line)


def main(args):
    for arg in args:
        sample = SAMPLES / arg
        if sample.is_file():
            words = sample.read_text(encoding="utf-8").strip()
            print(arg + " holds: " + (words or "(nothing)"))
        elif arg.endswith(".txt"):
            print(arg + " does not exist")
    print(("runs: python3 -m wordfreq " + shlex.join(args)).rstrip())
    env = dict(os.environ, PYTHONPATH=str(ROOT), PYTHONDONTWRITEBYTECODE="1")
    done = subprocess.run(
        [sys.executable, "-m", "wordfreq", *args],
        cwd=SAMPLES, env=env, capture_output=True, text=True,
    )
    show("stdout", done.stdout)
    show("stderr", done.stderr)
    print("exit code: " + str(done.returncode))


main(sys.argv[1:])
