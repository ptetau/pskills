### proj/pdlc/changes/CH-0001-top-option/show.sh
````sh
#!/bin/sh
# Shows one wordfreq run for the visual review of CH-0001.
# Usage: sh show.sh [TEXT] -- [wordfreq arguments]
# With TEXT (also an empty one), the script writes TEXT to notes.txt in a scratch folder.
# Then it runs wordfreq in that folder and prints standard output, standard error,
# and the exit code.

ROOT=$(cd "$(dirname "$0")/../../.." && pwd)
work=$(mktemp -d "${TMPDIR:-/tmp}/wordfreq-show.XXXXXX") || exit 1
trap 'rm -rf "$work"' EXIT

if [ "$#" -gt 0 ] && [ "$1" != "--" ]; then
  printf '%s' "$1" > "$work/notes.txt"
  if [ -n "$1" ]; then
    echo "notes.txt holds: $1"
  else
    echo "notes.txt holds: (nothing)"
  fi
  shift
fi
[ "${1:-}" = "--" ] && shift

shown="wordfreq"
for arg in "$@"; do
  case $arg in
    ''|*[!A-Za-z0-9_./+=:@%-]*) shown="$shown \"$arg\"" ;;
    *) shown="$shown $arg" ;;
  esac
done
echo "\$ $shown"

cd "$work" || exit 1
PYTHONDONTWRITEBYTECODE=1 PYTHONPATH="$ROOT" python3 -m wordfreq "$@" >"$work/.out" 2>"$work/.err"
code=$?

for stream in out err; do
  name=stdout
  [ "$stream" = err ] && name=stderr
  if [ -s "$work/.$stream" ]; then
    echo "$name:"
    sed 's/^/  /' "$work/.$stream"
  else
    echo "$name: (empty)"
  fi
done
echo "exit code: $code"
````
