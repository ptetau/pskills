### pdlc/changes/CH-0001-top-option/show_command.py

```
"""Run `python3 -m wordfreq` with the given arguments, and show what it did.

The recorder runs this helper in each step of visual.json. The command runs in the
folder `samples/`, so a file name such as `counts.txt` is a plain relative path. The
helper shows what each named sample file holds. Then it shows the standard output, the
standard error, and the exit code of the command, each on its own lines.
"""
import os
import shlex
import subprocess
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
SAMPLES = os.path.join(HERE, "samples")
ROOT = os.path.abspath(os.path.join(HERE, "..", "..", ".."))


def quote(arg):
    return '""' if arg == "" else shlex.quote(arg)


def describe(name):
    path = os.path.join(SAMPLES, name)
    if not os.path.isfile(path):
        return name + " does not exist"
    with open(path, encoding="utf-8") as handle:
        text = handle.read().strip()
    return name + " holds: " + text if text else name + " is an empty file"


def show_stream(label, text):
    if not text:
        print(label + ": (empty)")
        return
    print(label + ":")
    for line in text.splitlines():
        print("  " + line)


def main(args):
    for name in args:
        if name.endswith(".txt"):
            print(describe(name))
    print("samples$ " + " ".join(["python3 -m wordfreq", *[quote(arg) for arg in args]]))
    env = dict(os.environ, PYTHONPATH=ROOT, PYTHONDONTWRITEBYTECODE="1")
    done = subprocess.run(
        [sys.executable, "-m", "wordfreq", *args],
        cwd=SAMPLES, env=env, capture_output=True, text=True,
    )
    show_stream("stdout", done.stdout)
    show_stream("stderr", done.stderr)
    print("exit code: " + str(done.returncode))


main(sys.argv[1:])
```
