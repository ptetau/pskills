### durations/duration.py

````
import re

_DURATION = re.compile(r"(?=.)(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?")


def parse_duration(text: str) -> int:
    """Convert a duration string such as "1h30m" to seconds, or raise ValueError."""
    match = _DURATION.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours, minutes, seconds = match.groups()
    return int(hours or 0) * 3600 + int(minutes or 0) * 60 + int(seconds or 0)
````

import unittest

from duration import parse_duration


class ParseValidDurationTest(unittest.TestCase):
    def test_valid_durations(self) -> None:
        self.assertEqual(parse_duration("1h30m"), 5400)
        self.assertEqual(parse_duration("45s"), 45)
        self.assertEqual(parse_duration("1h2m3s"), 3723)


class RejectInvalidDurationTest(unittest.TestCase):
    def test_invalid_durations_raise_value_error(self) -> None:
        for text in ["", "90", "1x", "30m1h"]:
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_error_message_shows_the_text(self) -> None:
        with self.assertRaises(ValueError) as context:
            parse_duration("1x")
        self.assertEqual(str(context.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
