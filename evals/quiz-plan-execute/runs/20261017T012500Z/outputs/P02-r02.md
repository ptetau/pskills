### durations/duration.py
````python
import re

_DURATION = re.compile(r"(?=\d)(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?")
_SECONDS_PER_UNIT = (3600, 60, 1)


def parse_duration(text: str) -> int:
    """Convert a duration such as "1h30m" to a number of seconds."""
    match = _DURATION.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    return sum(
        int(count) * seconds
        for count, seconds in zip(match.groups(), _SECONDS_PER_UNIT)
        if count is not None
    )
````
### durations/test_duration.py

import unittest

from duration import parse_duration


class ParseDurationTest(unittest.TestCase):
    def test_valid_durations(self) -> None:
        """Valid duration strings convert to the total number of seconds."""
        self.assertEqual(parse_duration("1h30m"), 5400)
        self.assertEqual(parse_duration("45s"), 45)
        self.assertEqual(parse_duration("1h2m3s"), 3723)

    def test_invalid_durations_raise_value_error(self) -> None:
        """Text that is not a valid duration raises ValueError."""
        for text in ("", "90", "1x", "30m1h"):
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_error_message_shows_the_text(self) -> None:
        """The message is "invalid duration: " followed by repr(text)."""
        with self.assertRaises(ValueError) as caught:
            parse_duration("1x")
        self.assertEqual(str(caught.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
