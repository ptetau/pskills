### textutils/slug.py
````python
import re


def slugify(text: str, max_length: int | None = None) -> str:
    """Return a URL-safe slug of the text, cut to max_length if it is set."""
    slug = re.sub(r"[^a-z0-9]+", "-", text.lower()).strip("-")
    if max_length is not None:
        slug = slug[:max_length].rstrip("-")
    return slug
````

import unittest

from slug import slugify


class SlugifyTest(unittest.TestCase):
    def test_punctuation_becomes_one_hyphen(self) -> None:
        self.assertEqual(slugify("Hello, World!"), "hello-world")

    def test_runs_and_edges_are_trimmed(self) -> None:
        self.assertEqual(slugify("  a  b "), "a-b")

    def test_max_length_cuts_the_slug(self) -> None:
        self.assertEqual(slugify("Hello big world", max_length=9), "hello-big")

    def test_max_length_removes_hyphen_at_the_end(self) -> None:
        self.assertEqual(slugify("Hello big world", max_length=6), "hello")


if __name__ == "__main__":
    unittest.main()
