### textutils/slug.py
````python
"""Make URL-safe slugs from text."""

import re

_NOT_ASCII_ALNUM = re.compile(r"[^a-z0-9]+")


def slugify(text: str, max_length: int | None = None) -> str:
    """Return text as a lowercase ASCII slug, cut to max_length characters if set."""
    slug = _NOT_ASCII_ALNUM.sub("-", text.lower()).strip("-")
    if max_length is not None:
        slug = slug[:max_length].removesuffix("-")
    return slug
````

### textutils/test_slug.py

"""Unit tests for slug.slugify."""

import unittest

from slug import slugify


class SlugifyTest(unittest.TestCase):
    def test_replaces_each_run_of_other_characters_with_one_hyphen(self):
        self.assertEqual(slugify("Hello, World!"), "hello-world")

    def test_removes_hyphens_at_start_and_end(self):
        self.assertEqual(slugify("  a  b "), "a-b")

    def test_max_length_cuts_slug(self):
        self.assertEqual(slugify("Hello big world", max_length=9), "hello-big")

    def test_max_length_removes_hyphen_at_end_of_cut(self):
        self.assertEqual(slugify("Hello big world", max_length=6), "hello")


if __name__ == "__main__":
    unittest.main()
