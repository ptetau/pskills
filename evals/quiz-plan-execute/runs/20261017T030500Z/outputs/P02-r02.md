### durations/duration.py
````python
"""Convert duration strings such as "1h30m" to seconds."""

import re

_DURATION_RE = re.compile(
    r"(?:(?P<hours>[0-9]+)h)?"
    r"(?:(?P<minutes>[0-9]+)m)?"
    r"(?:(?P<seconds>[0-9]+)s)?"
)


def parse_duration(text: str) -> int:
    """Return the number of seconds in a duration such as "1h30m"."""
    parts = _DURATION_RE.fullmatch(text)
    # Each part is optional, so the pattern also matches "". Reject it here.
    if not text or parts is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(parts["hours"] or 0)
    minutes = int(parts["minutes"] or 0)
    seconds = int(parts["seconds"] or 0)
    return hours * 3600 + minutes * 60 + seconds
````

"""Unit tests for parse_duration."""

import unittest

from duration import parse_duration


class ParseValidDurationTest(unittest.TestCase):
    def test_valid_durations(self):
        self.assertEqual(parse_duration("1h30m"), 5400)
        self.assertEqual(parse_duration("45s"), 45)
        self.assertEqual(parse_duration("1h2m3s"), 3723)


class RejectBadInputTest(unittest.TestCase):
    def test_invalid_durations_raise_value_error(self):
        for text in ["", "90", "1x", "30m1h"]:
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_error_message_names_the_input(self):
        with self.assertRaises(ValueError) as caught:
            parse_duration("1x")
        self.assertEqual(str(caught.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
