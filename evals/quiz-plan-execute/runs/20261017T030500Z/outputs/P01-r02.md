### textutils/slug.py
````python
"""Make URL-safe slugs from text."""

import re

_NOT_ASCII_LETTER_OR_DIGIT = re.compile(r"[^a-z0-9]+")


def slugify(text: str, max_length: int | None = None) -> str:
    """Return a lowercase URL-safe slug of text, at most max_length characters if set."""
    slug = _NOT_ASCII_LETTER_OR_DIGIT.sub("-", text.lower()).strip("-")
    if max_length is not None:
        slug = slug[:max_length].rstrip("-")
    return slug
````

import unittest

from slug import slugify


class SlugifyTest(unittest.TestCase):
    def test_punctuation_run_becomes_one_hyphen(self):
        self.assertEqual(slugify("Hello, World!"), "hello-world")

    def test_spaces_collapse_and_end_hyphens_are_removed(self):
        self.assertEqual(slugify("  a  b "), "a-b")

    def test_max_length_cuts_slug(self):
        self.assertEqual(slugify("Hello big world", max_length=9), "hello-big")

    def test_max_length_removes_end_hyphen_after_cut(self):
        self.assertEqual(slugify("Hello big world", max_length=6), "hello")


if __name__ == "__main__":
    unittest.main()
