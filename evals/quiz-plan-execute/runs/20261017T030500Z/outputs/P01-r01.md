### textutils/slug.py
```python
"""Make URL-safe slugs from text."""

import re

# One run of characters that are not lowercase ASCII letters or digits.
_NON_ALPHANUMERIC_RUN = re.compile(r"[^a-z0-9]+")


def slugify(text: str, max_length: int | None = None) -> str:
    """Return a lowercase hyphen slug of text, at most max_length characters when it is set."""
    slug = _NON_ALPHANUMERIC_RUN.sub("-", text.lower()).strip("-")
    if max_length is not None:
        slug = slug[:max_length].removesuffix("-")
    return slug
```

### textutils/test_slug.py
```python
"""Unit tests for textutils/slug.py."""

import unittest

from slug import slugify


class SlugifyTest(unittest.TestCase):
    """Tests for slugify."""

    def test_basic_slug(self) -> None:
        """Lowercase the text, join the words with one hyphen, and trim the ends."""
        self.assertEqual(slugify("Hello, World!"), "hello-world")
        self.assertEqual(slugify("  a  b "), "a-b")

    def test_max_length(self) -> None:
        """Cut the slug to at most max_length characters, then remove a hyphen at the end."""
        self.assertEqual(slugify("Hello big world", max_length=9), "hello-big")
        self.assertEqual(slugify("Hello big world", max_length=6), "hello")


if __name__ == "__main__":
    unittest.main()
```
