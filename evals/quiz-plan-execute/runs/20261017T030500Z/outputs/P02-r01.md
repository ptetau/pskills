````python
"""Convert duration strings such as "1h30m" to a number of seconds."""

import re

_DURATION_RE = re.compile(r"(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?")


def parse_duration(text: str) -> int:
    """Return the number of seconds in a duration string such as "1h30m"."""
    match = _DURATION_RE.fullmatch(text)
    parts = match.groups() if match else ()
    if not any(parts):
        raise ValueError(f"invalid duration: {text!r}")
    hours, minutes, seconds = (int(part or 0) for part in parts)
    return hours * 3600 + minutes * 60 + seconds
````

````python
"""Unit tests for parse_duration."""

import unittest

from duration import parse_duration


class ParseDurationTest(unittest.TestCase):
    def test_parses_valid_durations(self) -> None:
        self.assertEqual(parse_duration("1h30m"), 5400)
        self.assertEqual(parse_duration("45s"), 45)
        self.assertEqual(parse_duration("1h2m3s"), 3723)

    def test_rejects_invalid_durations(self) -> None:
        for text in ["", "90", "1x", "30m1h"]:
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_error_message_names_the_input(self) -> None:
        with self.assertRaises(ValueError) as caught:
            parse_duration("1x")
        self.assertEqual(str(caught.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
````
