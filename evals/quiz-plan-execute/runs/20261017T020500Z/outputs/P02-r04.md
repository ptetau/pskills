### durations/duration.py
````python
"""Convert duration strings such as "1h30m" to a number of seconds."""

import re

_DURATION = re.compile(r"(?:(?P<h>\d+)h)?(?:(?P<m>\d+)m)?(?:(?P<s>\d+)s)?")
_UNIT_SECONDS = {"h": 3600, "m": 60, "s": 1}


def parse_duration(text: str) -> int:
    """Return the number of seconds in a duration string such as "1h30m"."""
    match = _DURATION.fullmatch(text)
    if not text or match is None:
        raise ValueError(f"invalid duration: {text!r}")
    amounts = match.groupdict(default="0")
    return sum(int(amounts[unit]) * size for unit, size in _UNIT_SECONDS.items())
````

````python
"""Unit tests for parse_duration."""

import unittest

from duration import parse_duration


class ParseDurationTest(unittest.TestCase):
    """Tests for parse_duration."""

    def test_valid_durations(self) -> None:
        """Valid duration strings convert to the correct number of seconds."""
        cases = {"1h30m": 5400, "45s": 45, "1h2m3s": 3723}
        for text, expected in cases.items():
            with self.subTest(text=text):
                self.assertEqual(parse_duration(text), expected)

    def test_invalid_durations(self) -> None:
        """Text that is not a valid duration raises ValueError."""
        for text in ("", "90", "1x", "30m1h"):
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_invalid_duration_message(self) -> None:
        """The error message gives the repr of the bad text."""
        with self.assertRaises(ValueError) as context:
            parse_duration("1x")
        self.assertEqual(str(context.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
````
