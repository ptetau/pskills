### textutils/slug.py
````python
"""Make URL-safe slugs from text."""

import re

_NOT_ALNUM = re.compile(r"[^a-z0-9]+")


def slugify(text: str, max_length: int | None = None) -> str:
    """Return a hyphenated lowercase ASCII slug, cut to max_length if it is set."""
    slug = _NOT_ALNUM.sub("-", text.lower()).strip("-")
    if max_length is not None:
        slug = slug[:max_length].removesuffix("-")
    return slug
````

"""Tests for the slugify helper."""

import unittest

from slug import slugify


class SlugifyTest(unittest.TestCase):
    def test_basic(self) -> None:
        self.assertEqual(slugify("Hello, World!"), "hello-world")
        self.assertEqual(slugify("  a  b "), "a-b")

    def test_max_length(self) -> None:
        self.assertEqual(slugify("Hello big world", max_length=9), "hello-big")
        self.assertEqual(slugify("Hello big world", max_length=6), "hello")


if __name__ == "__main__":
    unittest.main()
