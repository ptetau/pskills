"""Convert duration strings such as "1h30m" to a number of seconds."""

import re

_DURATION = re.compile(r"(?:(?P<h>\d+)h)?(?:(?P<m>\d+)m)?(?:(?P<s>\d+)s)?")
_SECONDS_PER_UNIT = {"h": 3600, "m": 60, "s": 1}


def parse_duration(text: str) -> int:
    """Return the number of seconds in a duration string such as "1h30m"."""
    parts = _DURATION.fullmatch(text).groupdict()
    return sum(_SECONDS_PER_UNIT[unit] * int(value) for unit, value in parts.items() if value)

### durations/duration.py
````python
"""Convert duration strings such as "1h30m" to a number of seconds."""

import re

_DURATION = re.compile(r"(?:(?P<h>\d+)h)?(?:(?P<m>\d+)m)?(?:(?P<s>\d+)s)?")
_SECONDS_PER_UNIT = {"h": 3600, "m": 60, "s": 1}


def parse_duration(text: str) -> int:
    """Return the seconds in a duration string such as "1h30m", or raise ValueError."""
    match = _DURATION.fullmatch(text)
    if not text or match is None:
        raise ValueError(f"invalid duration: {text!r}")
    parts = match.groupdict()
    return sum(_SECONDS_PER_UNIT[unit] * int(value) for unit, value in parts.items() if value)
````

"""Unit tests for parse_duration."""

import unittest

from duration import parse_duration


class ParseDurationTest(unittest.TestCase):
    def test_valid_durations(self):
        self.assertEqual(parse_duration("1h30m"), 5400)
        self.assertEqual(parse_duration("45s"), 45)
        self.assertEqual(parse_duration("1h2m3s"), 3723)

    def test_invalid_durations_raise_value_error(self):
        for text in ("", "90", "1x", "30m1h"):
            with self.subTest(text=text):
                with self.assertRaises(ValueError) as caught:
                    parse_duration(text)
                self.assertEqual(str(caught.exception), f"invalid duration: {text!r}")

    def test_error_message_for_unknown_unit(self):
        with self.assertRaises(ValueError) as caught:
            parse_duration("1x")
        self.assertEqual(str(caught.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
