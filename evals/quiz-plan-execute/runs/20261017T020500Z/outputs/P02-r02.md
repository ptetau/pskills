### durations/duration.py
````python
"""Convert duration strings such as "1h30m" to a number of seconds."""

import re

_DURATION_RE = re.compile(r"(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?")
_SECONDS_PER_HOUR = 3600
_SECONDS_PER_MINUTE = 60


def parse_duration(text: str) -> int:
    """Return the number of seconds in a duration string such as "1h30m"."""
    match = _DURATION_RE.fullmatch(text)
    # Each part of the pattern is optional, so the pattern also matches "".
    if not text or match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours, minutes, seconds = match.groups()
    return (
        int(hours or 0) * _SECONDS_PER_HOUR
        + int(minutes or 0) * _SECONDS_PER_MINUTE
        + int(seconds or 0)
    )
````

### durations/test_duration.py

````python
"""Unit tests for parse_duration."""

import unittest

from duration import parse_duration


class ParseDurationTest(unittest.TestCase):
    def test_valid_durations(self) -> None:
        cases = {"1h30m": 5400, "45s": 45, "1h2m3s": 3723}
        for text, seconds in cases.items():
            with self.subTest(text=text):
                self.assertEqual(parse_duration(text), seconds)

    def test_invalid_durations_raise_value_error(self) -> None:
        for text in ["", "90", "1x", "30m1h"]:
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_error_message_shows_the_text(self) -> None:
        with self.assertRaises(ValueError) as caught:
            parse_duration("1x")
        self.assertEqual(str(caught.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
````
