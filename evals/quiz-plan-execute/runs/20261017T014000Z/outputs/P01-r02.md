### textutils/slug.py
````python
import re

_NON_ALPHANUMERIC_RUN = re.compile(r"[^a-z0-9]+")


def slugify(text: str, max_length: int | None = None) -> str:
    """Return the URL-safe slug of text, cut to max_length characters when it is set."""
    slug = _NON_ALPHANUMERIC_RUN.sub("-", text.lower()).strip("-")
    if max_length is not None:
        slug = slug[:max_length].rstrip("-")
    return slug
````
### textutils/test_slug.py
````python
import unittest

from slug import slugify


class SlugifyTest(unittest.TestCase):
    """Tests for slugify."""

    def test_punctuation_becomes_one_hyphen(self) -> None:
        """Punctuation and spaces between words become one hyphen."""
        self.assertEqual(slugify("Hello, World!"), "hello-world")

    def test_edge_whitespace_and_runs_are_trimmed(self) -> None:
        """Runs of spaces collapse, and edge hyphens are removed."""
        self.assertEqual(slugify("  a  b "), "a-b")

    def test_max_length_cuts_the_slug(self) -> None:
        """A slug longer than max_length is cut to max_length characters."""
        self.assertEqual(slugify("Hello big world", max_length=9), "hello-big")

    def test_max_length_removes_the_hyphen_at_the_end(self) -> None:
        """A cut that ends on a hyphen loses that hyphen."""
        self.assertEqual(slugify("Hello big world", max_length=6), "hello")


if __name__ == "__main__":
    unittest.main()
````
