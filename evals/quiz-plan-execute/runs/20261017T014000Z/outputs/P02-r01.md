````
"""Convert duration strings such as "1h30m" to a number of seconds."""

import re

_DURATION_RE = re.compile(r"(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?")


def parse_duration(text: str) -> int:
    """Return the number of seconds in a duration string such as "1h30m"."""
    match = _DURATION_RE.fullmatch(text)
    if not text or match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours, minutes, seconds = match.groups(default="0")
    return int(hours) * 3600 + int(minutes) * 60 + int(seconds)
````

````
"""Unit tests for duration parsing."""

import unittest

from duration import parse_duration


class ParseValidDurationTest(unittest.TestCase):
    """Tests for valid duration strings."""

    def test_hours_and_minutes(self) -> None:
        """Convert hours and minutes to seconds."""
        self.assertEqual(parse_duration("1h30m"), 5400)

    def test_seconds_only(self) -> None:
        """Convert seconds alone to seconds."""
        self.assertEqual(parse_duration("45s"), 45)

    def test_all_three_units(self) -> None:
        """Convert hours, minutes and seconds to seconds."""
        self.assertEqual(parse_duration("1h2m3s"), 3723)


class ParseInvalidDurationTest(unittest.TestCase):
    """Tests for text that is not a valid duration."""

    def test_rejects_invalid_text(self) -> None:
        """Raise ValueError for each kind of invalid text."""
        for text in ("", "90", "1x", "30m1h"):
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_message_shows_the_text(self) -> None:
        """Put the repr of the text in the error message."""
        with self.assertRaises(ValueError) as caught:
            parse_duration("1x")
        self.assertEqual(str(caught.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
````
