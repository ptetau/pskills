### durations/duration.py
````python
import re

_DURATION = re.compile(r"(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?")
_SECONDS_PER_UNIT = (3600, 60, 1)


def parse_duration(text: str) -> int:
    """Convert a duration string such as "1h30m" to a number of seconds."""
    match = _DURATION.fullmatch(text)
    if not text or match is None:
        raise ValueError(f"invalid duration: {text!r}")
    return sum(
        int(digits) * unit
        for digits, unit in zip(match.groups(), _SECONDS_PER_UNIT)
        if digits
    )
````

````python
import unittest

from duration import parse_duration


class ParseDurationTest(unittest.TestCase):
    def test_valid_durations(self) -> None:
        """Valid duration strings convert to the right number of seconds."""
        cases = {
            "1h30m": 5400,
            "45s": 45,
            "1h2m3s": 3723,
        }
        for text, seconds in cases.items():
            with self.subTest(text=text):
                self.assertEqual(parse_duration(text), seconds)

    def test_invalid_durations_raise_value_error(self) -> None:
        """Text that is not a valid duration raises ValueError."""
        for text in ["", "90", "1x", "30m1h"]:
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    parse_duration(text)

    def test_invalid_duration_message(self) -> None:
        """The error message names the bad text with repr."""
        with self.assertRaises(ValueError) as context:
            parse_duration("1x")
        self.assertEqual(str(context.exception), "invalid duration: '1x'")


if __name__ == "__main__":
    unittest.main()
````
