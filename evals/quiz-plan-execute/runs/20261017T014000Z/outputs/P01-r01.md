### textutils/slug.py
````python
import re

_NON_ALPHANUMERIC = re.compile(r"[^a-z0-9]+")


def slugify(text: str, max_length: int | None = None) -> str:
    """Return a URL-safe slug of the text, cut to max_length characters if it is set."""
    slug = _NON_ALPHANUMERIC.sub("-", text.lower()).strip("-")
    if max_length is not None:
        slug = slug[:max_length].rstrip("-")
    return slug
````
### textutils/test_slug.py
````python
import unittest

from slug import slugify


class TestSlugify(unittest.TestCase):
    def test_basic(self):
        self.assertEqual(slugify("Hello, World!"), "hello-world")
        self.assertEqual(slugify("  a  b "), "a-b")

    def test_max_length(self):
        self.assertEqual(slugify("Hello big world", max_length=9), "hello-big")
        self.assertEqual(slugify("Hello big world", max_length=6), "hello")


if __name__ == "__main__":
    unittest.main()
````
